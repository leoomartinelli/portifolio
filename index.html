<?php

final class Leonardo
{
    private int $idade = 18;
    private string $empresa = 'DATACUBE';
    private array $stack = ['PHP', 'JavaScript'];

    public function apresentar(): string
    {
        return "{$this->idade} anos, trainee "
             . "na {$this->empresa}. "
             . implode(' + ', $this->stack);
    }
}
